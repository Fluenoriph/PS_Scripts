BEGIN;

CREATE TABLE file_type (
file_type_id SERIAL PRIMARY KEY,
file_type_name TEXT NOT NULL UNIQUE
);

INSERT INTO file_type (file_type_name) VALUES 
('APE'),
('FLAC'),
('M4A'),
('MP3'),
('MUSEPACK'),
('WAV');

CREATE TABLE import_batch (
batch_id SERIAL PRIMARY KEY,
created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
description TEXT
);

CREATE TABLE file_data (
file_data_id SERIAL PRIMARY KEY,
file_location TEXT,
file_name TEXT NOT NULL,
file_size_bytes BIGINT NOT NULL,
file_hash TEXT NOT NULL UNIQUE,
file_type_id INTEGER NOT NULL,
batch_id INTEGER NOT NULL,
FOREIGN KEY (file_type_id) REFERENCES file_type (file_type_id) ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY (batch_id) REFERENCES import_batch (batch_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE audio_metadata (
metadata_id SERIAL PRIMARY KEY,
artist TEXT,
title TEXT,
album TEXT,
genre TEXT,
year SMALLINT,
file_data_id INTEGER NOT NULL,
FOREIGN KEY (file_data_id) REFERENCES file_data (file_data_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE audio_stream_data (
stream_data_id SERIAL PRIMARY KEY,
codec TEXT NOT NULL,
bitrate_kbps INTEGER,
sample_rate_khz INTEGER,
duration_seconds REAL, -- Тип REAL, так как mutagen выдает не целые значения.
bits_per_sample_bit INTEGER,
channels TEXT,
file_data_id INTEGER NOT NULL,
FOREIGN KEY (file_data_id) REFERENCES file_data (file_data_id) ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE INDEX idx_file_hash ON file_data(file_hash);
CREATE INDEX idx_artist ON audio_metadata(artist);
CREATE INDEX idx_title ON audio_metadata(title);
CREATE INDEX idx_genre ON audio_metadata(genre);
CREATE INDEX idx_year ON audio_metadata(year);

COMMIT;

-- Find duplicates

SELECT file_location, file_name, file_size_mb FROM file_data WHERE file_hash IN (SELECT file_hash FROM file_data GROUP BY file_hash HAVING COUNT(*) > 1) ORDER BY file_size_mb; -- not index: 4,499 ms

-- Duplicate sum

SELECT (SELECT COUNT(file_hash) FROM file_data) - (SELECT COUNT(DISTINCT file_hash) FROM file_data);

-- SELECT to album name

SELECT f.file_name, f.file_size_mb, t.file_type_name, f.file_hash FROM file_data f JOIN file_type t ON f.file_type_id = t.file_type_id
JOIN audio_metadata a ON f.file_data_id = a.file_data_id WHERE a.album = 'album name';
