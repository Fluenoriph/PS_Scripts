CREATE TABLE Calculator_Type (
calc_type_id INTEGER PRIMARY KEY,
calc_type_name TEXT NOT NULL UNIQUE
);

CREATE TABLE Registration_Time (
reg_time_id SERIAL PRIMARY KEY,
reg_time_value TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
calc_type_id INTEGER NOT NULL,
FOREIGN KEY (calc_type_id) REFERENCES calculator_type (calc_type_id) ON DELETE CASCADE
);

CREATE TABLE Formula_Type_Calculator_Data (
data_id SERIAL PRIMARY KEY,
parameter REAL[],
result REAL[],
reg_time_id INTEGER NOT NULL,
FOREIGN KEY (reg_time_id) REFERENCES registration_time (reg_time_id) ON DELETE CASCADE
);

CREATE TABLE Noise_Calculator_Data (
data_id SERIAL PRIMARY KEY,
source_level REAL[],
background_level REAL[],
delta REAL[],
result REAL[],
reg_time_id INTEGER NOT NULL,
FOREIGN KEY (reg_time_id) REFERENCES registration_time (reg_time_id) ON DELETE CASCADE
);
